// Helper functions
function _g($p) { 
    $real = realpath($p);
    if($real && strpos($real, _B_) === 0) return $real;
    return false;
}

function _p($f) { 
    return substr(sprintf('%o', fileperms($f)), -4); 
}

function _pm($f) {
    $m = fileperms($f); 
    $s = ($m & 0x4000) ? 'd' : '-';
    $x = [0x0100=>'r', 0x0080=>'w', 0x0040=>'x', 0x0020=>'r', 0x0010=>'w', 0x0008=>'x', 
          0x0004=>'r', 0x0002=>'w', 0x0001=>'x'];
    foreach($x as $b=>$c) { 
        $s .= ($m & $b) ? $c : '-'; 
    }
    return $s;
}

function _go() { 
    header('Location: ?_r='.urlencode(_B_)); 
    exit; 
}

function format_size($bytes) {
    if ($bytes >= 1073741824) return number_format($bytes / 1073741824, 2) . ' GB';
    if ($bytes >= 1048576) return number_format($bytes / 1048576, 2) . ' MB';
    if ($bytes >= 1024) return number_format($bytes / 1024, 2) . ' KB';
    return $bytes . ' B';
}
